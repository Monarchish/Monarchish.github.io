# 编辑手工提单

Open "台账"

Ctrl + C "台账" - "捆包号（S）"

Ctrl + C "台账" - "品名（AA）"

Ctrl + C "台账" - "产地（AB）"

Ctrl + C "台账" - "牌号（AC）"

Ctrl + C "台账" - "规格（AD）"

Open "手工提单"

Ctrl + V "手工提单" - "捆包号/卡号（A）"

Open "台账"

Ctrl + C "台账" - "吨位（T）"

Open "手工提单"

Ctrl + V "手工提单" - "重量（吨）（H）"

Write "手工提单" - "数量（G）"

（1）

Ctrl + D "手工提单" - "数量（G）"

# 泉冠库需要转库函怎么办

Open "欧冶云仓" - "物流管理" - "云仓库存"

Write "云仓库存" - "所在仓库"

（上海宝钢运输有限公司15号库（泉冠库））

Left Click "云仓库存" - "查询"

Write "云仓库存" - "10条/页"

（50条/页）

Left Click "云仓库存" - "全选"

Left Click "云仓库存" - "选择导出"

Open "云仓库存"

Ctrl + C "捆包号（E）"

Open "台账"

Ctrl + V "台账" - "（BX）"

Filter "台账" - "（BW）"

（0）

Ctrl + C "钢厂资源号（AF）"

Open "出库函"

Ctrl + V "资源号（A）"

Open "台账"

Ctrl + C "台账" - "捆包号（S）"

Ctrl + C "台账" - "品名（AA）"

Ctrl + C "台账" - "牌号（AC）"

Ctrl + C "台账" - "规格（AD）"

Open "出库函"

Ctrl + V "捆包号（B）"

Open "台账"

Ctrl + C "台账" - "吨位（T）"

Open "出库函"

Ctrl + V "重量(吨)（F）"

Ctrl + S

Ctrl + Alt + P

Ctrl + Alt + A

（发给泉冠库。）

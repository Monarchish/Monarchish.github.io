async function loadContent(tab, pageId) { const loader = document.getElementById('contentLoader'); // 判断文件路径 let filePath; if (tab === 'zhunru') { // 准入分类：从 content/准入/ 文件夹加载 filePath = `content/准入/${pageId}.html`; } else { // 其他分类：直接从 content/ 文件夹加载 filePath = `content/${tab}.html`; } try { const resp = await fetch(filePath); if (!resp.ok) { loader.innerHTML = `

页面加载失败，请刷新重试。

`; return; } const html = await resp.text(); loader.innerHTML = html; generateTOC(); document.getElementById('contentArea').scrollTop = 0; window.scrollTo({ top: 0, behavior: 'smooth' }); } catch (e) { console.error('内容加载失败:', e); loader.innerHTML = `

内容加载失败，请刷新重试。

`; } }
